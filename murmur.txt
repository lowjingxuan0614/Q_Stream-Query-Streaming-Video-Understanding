 function runProcessingPipeline() {
            if (!isStreaming) return;

            // 1. Streaming Frame Sampling (1.2.2)
            frameCount++;
            addLog(`Sampled frame #${frameCount} (fixed rate)`, 'log-frame');
            
            // 2. Frame Encoding (1.2.3)
            const embedding = `emb_${frameCount}_${Math.random().toString(16).slice(2,8)}`;
            addLog(`Encoded frame #${frameCount} → embedding: ${embedding}`, 'log-encode');

            // 3. Event Construction (1.2.4)
            if (frameCount % 5 === 0) {
                const eventStart = frameCount - 4;
                addLog(`New event detected: [${eventStart} → ${frameCount}] (time window)`, 'log-event');
            }

            // 4. Query Forecasting (1.2.5)
            if (frameCount % 8 === 0) {
                addLog(`Forecasted query types: What (action), When (timing), How (steps)`, 'log-query');
            }

            // 5. Query-Aware Memory Management (1.2.6)
            if (frameCount % 10 === 0) {
                addLog(`Memory updated: Kept important events | Merged low-relevance events`, 'log-memory');
            }

            // Update progress
            updateProgress(Math.min(100, (frameCount / 100) * 100));

            // Repeat every 1 second (simulate real-time stream)
            setTimeout(runProcessingPipeline, 1000);
        }
